create or replace function public.unlock_chapter_with_wallet(p_chapter_id uuid) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_user_id uuid := auth.uid(); v_price bigint; v_title text; v_balance bigint; begin if v_user_id is null then raise exception 'Chưa đăng nhập'; end if; if exists (select 1 from public.chapter_unlocks where user_id = v_user_id and chapter_id = p_chapter_id) then return jsonb_build_object('status', 'already_unlocked'); end if; select price, title into v_price, v_title from public.chapters where id = p_chapter_id; if not found then raise exception 'Không tìm thấy chương'; end if; select balance into v_balance from public.wallets where user_id = v_user_id for update; if v_balance is null or v_balance < v_price then raise exception 'Số dư ví không đủ'; end if; update public.wallets set balance = balance - v_price, updated_at = now() where user_id = v_user_id; insert into public.chapter_unlocks (user_id, chapter_id, price_paid) values (v_user_id, p_chapter_id, v_price); insert into public.wallet_transactions (user_id, type, amount, balance_after, description) values (v_user_id, 'CHAPTER_UNLOCK', -v_price, v_balance - v_price, 'Mở khóa ' || v_title); update public.chapters set purchase_count = purchase_count + 1 where id = p_chapter_id; return jsonb_build_object('status', 'unlocked', 'balance', v_balance - v_price); end; $$;
